تطبيق مفتاحي الكلية - شرح المعمارية والأكواد البرمجية سطرًا بسطر

الموضوع: توثيق الشرح البرمجي الكامل لتطبيق استعادة المفاتيح المفقودة في الكلية (مثال: قاعة 202) والتنسيق المباشر مع رجال الأمن والمباني.

التقنيات المستخدمة: React 18 + Tailwind CSS + Web Audio API + LocalStorage API

🔊 2. كلاس الصوتيات والتنبيهات البرمجية (src/utils/audio.js)

يعتمد الملف على Web Audio API لتوليد أصوات التنبيه دون الحاجة لملفات صوت خارجية:

// src/utils/audio.js

class SoundEffects {
  constructor() {
    this.ctx = null;     // سياق الصوت المعرف من المتصفح
    this.enabled = true; // مفتاح تفعيل/تعطيل الصوت
  }

  // دالة تهيئة سياق الصوت
  initContext() {
    if (!this.ctx && typeof window !== 'undefined') {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume();
    }
  }

  // دالة تشغيل صوت الإنذار الأمني (عند الإبلاغ عن مفتاح مفقود بقاعة 202)
  playAlertSound() {
    if (!this.enabled) return;
    this.initContext();
    if (!this.ctx) return;

    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = 'sawtooth'; // موجة مشرشرة للإنذار
    osc.frequency.setValueAtTime(880, now);          // نغمة A5 (880Hz)
    osc.frequency.setValueAtTime(1174.66, now + 0.15); // نغمة D6 (1174Hz)
    osc.frequency.setValueAtTime(880, now + 0.3);
    osc.frequency.setValueAtTime(1174.66, now + 0.45);

    gain.gain.setValueAtTime(0.3, now);
    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.6); // إيقاف الصوت بعد 0.6 ثانية
  }

  // دالة تشغيل صوت النجاح (عند تأكيد مكان العثور على المفتاح)
  playSuccessSound() {
    if (!this.enabled) return;
    this.initContext();
    if (!this.ctx) return;

    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = 'sine';
    osc.frequency.setValueAtTime(523.25, now);       // C5
    osc.frequency.setValueAtTime(659.25, now + 0.1); // E5
    osc.frequency.setValueAtTime(783.99, now + 0.2); // G5
    osc.frequency.setValueAtTime(1046.50, now + 0.3);// C6 (سلم نغمات صاعد)

    gain.gain.setValueAtTime(0.2, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.6);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.6);
  }
}

export const sounds = new SoundEffects();

💾 3. دوال التخزين التلقائي (src/utils/storage.js)

// src/utils/storage.js

const STORAGE_KEY = 'college_keys_reports_v1';

// قراءة البلاغات المخزنة أو إرجاع البلاغ الافتراضي لقاعة 202
export function getInitialReports() {
  const saved = localStorage.getItem(STORAGE_KEY);
  if (saved) {
    try { return JSON.parse(saved); } catch (e) { console.error(e); }
  }
  return [{
    id: 'KEY-2026-001',
    type: 'lost',
    title: 'مفتاح قاعة 202 (مبنى أ - الهندسة)',
    building: 'مبنى أ - الهندسة',
    room: 'قاعة 202',
    description: 'تم فقدان المفتاح الخاص بالقاعة بعد نهاية محاضرة البرمجة.',
    keychainTag: 'ميدالية سوداء برقم 202',
    status: 'searching',
    reporterName: 'د. عبد الله أحمد',
    dateLost: new Date().toISOString(),
    foundLocation: null,
    messages: []
  }];
}

// حفظ البلاغات في LocalStorage
export function saveReports(reports) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(reports));
}

// تحديث موقع المفتاح المكتشف برمجياً
export function resolveReportLocationInStorage(reportId, status, location, finderName) {
  const reports = getInitialReports();
  const updated = reports.map(r => {
    if (r.id === reportId) {
      return {
        ...r,
        status: status,           // تحويل الحالة إلى (found)
        foundLocation: location, // كتابة النص المحدد للموقع
        foundBy: finderName || r.foundBy
      };
    }
    return r;
  });
  saveReports(updated);
  return updated;
}

// src/App.jsx (مقتطف منطق العرض والفلترة وبنر قاعة 202)

function MainAppContent() {
  const { reports, searchQuery, selectedRoom, setChatReport } = useApp();

  // تصفية البلاغات حسب البحث ورقم القاعة
  const filteredReports = reports.filter(r => {
    if (searchQuery.trim()) {
      const q = searchQuery.toLowerCase();
      if (!r.room?.toLowerCase().includes(q) && !r.title?.toLowerCase().includes(q)) return false;
    }
    if (selectedRoom !== 'ALL' && !r.room?.includes(selectedRoom)) return false;
    return true;
  });

  // استخراج بلاغ قاعة 202 بشكل خاص لإظهاره في البنر العلوي
  const room202Report = reports.find(r => r.room && r.room.includes('202'));

  return (
    <div className="min-h-screen bg-slate-950 text-slate-100">
      <Navbar />

      <main className="max-w-7xl mx-auto px-4 py-8 space-y-8">
        
        {/* بنر الطوارئ المباشر لقاعة 202 */}
        {room202Report && (
          <div className="glass-panel-accent rounded-3xl p-6 border border-cyan-400/40">
            <div className="flex items-center justify-between">
              <div>
                <span className="bg-cyan-500 text-slate-950 font-black text-[10px] px-2.5 py-0.5 rounded-full">
                  بلاغ عاجل متفاعل 🔑
                </span>
                <h2 className="text-base font-extrabold text-white mt-1">
                  حالة مفتاح قاعة 202 (مبنى أ - الهندسة)
                </h2>
                <p className="text-xs text-slate-300 mt-1">
                  {room202Report.status === 'found' ? (
                    <span className="text-emerald-300 font-extrabold">
                      📍 تم العثور على المفتاح: {room202Report.foundLocation}
                    </span>
                  ) : (
                    <span>جاري البحث بواسطة رجال الأمن والكلية...</span>
                  )}
                </p>
              </div>

              <button onClick={() => setChatReport(room202Report)} className="btn-primary">
                فتح متابعة بلاغ قاعة 202 ←
              </button>
            </div>
          </div>
        )}

      </main>
    </div>
  );
}
