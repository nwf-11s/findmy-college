🔊 2. نظام التنبيهات الصوتية البرمجية (src/utils/audio.js)

يتميز المشروع باستخدام **Web Audio API** لتوليد نغمات صوتية برمجية ذات جودة عالية بدون الحاجة لتنزيل أي ملفات MP3 خارجية:

src/utils/audio.jsJavaScript
// src/utils/audio.js - نظام الأصوات والتنبيهات الصوتية

class SoundEffects {
  constructor() {
    this.ctx = null;     // سياق الصوت التابع للمتصفح
    this.enabled = true; // حالة مفتاح تفعيل/تعطيل الصوت
  }

  // تهيئة سياق الصوت (AudioContext) عند أول تفاعل من المستخدم
  initContext() {
    if (!this.ctx && typeof window !== 'undefined') {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume(); // استئناف الصوت إن كان متوقفاً
    }
  }

  // تشغيل صوت صفارة إنذار أمني عند فقدان المفتاح في قاعة 202
  playAlertSound() {
    if (!this.enabled) return;
    this.initContext();
    if (!this.ctx) return;

    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator(); // مولد التردد
    const gain = this.ctx.createGain();      // التحكم في مستوى الصوت

    osc.type = 'sawtooth'; // موجة مشرشرة للإنذار
    osc.frequency.setValueAtTime(880, now);          // نغمة A5
    osc.frequency.setValueAtTime(1174.66, now + 0.15); // نغمة D6
    osc.frequency.setValueAtTime(880, now + 0.3);
    osc.frequency.setValueAtTime(1174.66, now + 0.45);

    gain.gain.setValueAtTime(0.3, now);
    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.6); // إيقاف التشغيل بعد 0.6 ثانية
  }

  // نغمة نجاح العثور على المفتاح (سلم صاعد مبهج C5 -> C6)
  playSuccessSound() {
    if (!this.enabled) return;
    this.initContext();
    if (!this.ctx) return;

    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = 'sine';
    osc.frequency.setValueAtTime(523.25, now);       // C5
    osc.frequency.setValueAtTime(659.25, now + 0.1); // E5
    osc.frequency.setValueAtTime(783.99, now + 0.2); // G5
    osc.frequency.setValueAtTime(1046.50, now + 0.3);// C6

    gain.gain.setValueAtTime(0.2, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.6);

    osc.connect(gain);
    gain.connect(this.ctx.destination);

    osc.start(now);
    osc.stop(now + 0.6);
  }
}

export const sounds = new SoundEffects();

💾 3. دوال الحفظ والتخزين المحلية (src/utils/storage.js)

يضمن هذا الملف حفظ البيانات في ذاكرة المتصفح حتى لا تضيع البلاغات والمحادثات عند عمل Refresh:

src/utils/storage.jsJavaScript
// src/utils/storage.js - التخزين المحلي

const STORAGE_KEY = 'college_keys_reports_v1';

// قراءة البلاغات المخزنة أو إرجاع البلاغ الافتراضي لقاعة 202
export function getInitialReports() {
  const saved = localStorage.getItem(STORAGE_KEY);
  if (saved) {
    try { return JSON.parse(saved); } catch (e) { console.error(e); }
  }
  return [{
    id: 'KEY-2026-001',
    type: 'lost',
    title: 'مفتاح قاعة 202 (مبنى أ - الهندسة)',
    building: 'مبنى أ - الهندسة',
    room: 'قاعة 202',
    description: 'تم فقدان المفتاح الخاص بالقاعة بعد نهاية محاضرة البرمجة.',
    keychainTag: 'ميدالية سوداء برقم 202',
    status: 'searching',
    reporterName: 'د. عبد الله أحمد',
    dateLost: new Date().toISOString(),
    foundLocation: null,
    messages: []
  }];
}

// تحديث مكان العثور على المفتاح وتغيير حالته
export function resolveReportLocationInStorage(reportId, status, location, finderName) {
  const reports = getInitialReports();
  const updated = reports.map(r => {
    if (r.id === reportId) {
      return {
        ...r,
        status: status,           // تحويل الحالة إلى (found)
        foundLocation: location, // تسجيل النص المكتوب لمكان العثور
        foundBy: finderName || r.foundBy
      };
    }
    return r;
  });
  saveReports(updated);
  return updated;
}

  • ChatModal.jsx: يتيح المحادثة وتحديد مكان العثور مع نغمة النجاح.

🔄 7. دورة عمل التطبيق الشاملة

سير العملية بالتفصيل:
  1. يقوم المستخدم بالإبلاغ عن مفتاح مفقود بقاعة 202 بضغطة زر واحدة.
  2. يصدر المتصفح صوت إنذار صفارة أمنية سريعة محذراً رجال الأمن والقاعات المجاورة.
  3. يظهر بلاغ قاعة 202 في أعلى التطبيق ومميز باللون الأحمر على الخريطة التفاعلية.
  4. عندما يجد أحدهم المفتاح، يحدد موقعه في المحادثة -> يصدر صوت النجاح المبهج.
  5. يتحول البلاغ للون الأخضر ويظهر بنر "📍 أين تم العثور على المفتاح؟" لجميع المستخدمين.
