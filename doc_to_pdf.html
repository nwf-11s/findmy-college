نظام ذكي لاستعادة المفاتيح المفقودة في الكلية (مثال: قاعة 202) والتنسيق مع فريق الأمن والمباني
البيئة: React 18 + Tailwind CSS + Web Audio API
إعداد وتطوير: فريق الذكاء الاصطناعي البرمجي
التاريخ: سبتمبر 2026

1. المقدمة والفكرة العامة للمشروع

تم تطوير **تطبيق مفتاحي الكلية** كحل رقمي تفاعلي سريع ومخصص للجامعات والكليات لمواجهة مشكلة فقدان المفاتيح الخاصة بالقاعات الدراسية والمعامل (مثل قاعة 202). يربط التطبيق بين الطلاب، أعضاء الهيئة التدريسية، ورجال الأمن في منصة موحدة توفر تنبيهات أمنية صوتية وبصرية فورية.

💡 الميزات الرئيسية التي تم تنفيذها:
  • الإبلاغ السريع عن قاعة 202: زر مخصص للإبلاغ الفوري بضغطة زر واحدة.
  • نظام الصوت البرمجي (Web Audio API): توليد أصوات إنذار صفارة أمنية ونغمات العثور بنجاح دون الحاجة لملفات صوت خارجية.
  • تحديد موقع المفتاح المكتشف: إمكانية تحديد وإظهار بنر بارز "📍 أين تم العثور على المفتاح؟" لجميع المستخدمين.
  • محادثة تفاعلية فورية: تواصل مباشر بين المالك ومكتشف المفتاح أو رجال الأمن.
  • لوحة تحكم أمنية ومخطط تفاعلي: تتبع حالة قاعة 202 على الخريطة التفاعلية.

2. الهيكلية والتقنيات (Tech Stack)

التقنية / المكتبة الاستخدام الوظيفي
React 18 بناء واجهة المستخدم وتجزئة التطبيق إلى مكونات من الناحية الهيكلية.
React Context API إدارة الحالة المركزية للبلاغات والمحادثات والتبديل بين أدوار المستخدمين.
Web Audio API توليد الترددات والنغمات الصوتية التنبيهية فورية في المتصفح.
Tailwind CSS التصميم البصري، الألوان المتناسقة (Dark Mode)، والواجهات الزجاجية (Glassmorphism).
LocalStorage API حفظ وتخزين البيانات محلية لضمان استمرارية البلاغات والمحادثات.

3. نظام التنبيهات الصوتية البرمجية (src/utils/audio.js)

يعتمد التطبيق على كلاس SoundEffects لتوليد موجات صوتية نَقية لتنبيه المستخدِم برمجياً: 

// src/utils/audio.js - نظام التنبيهات الصوتية باستخدام Web Audio API

class SoundEffects {
  constructor() {
    this.ctx = null;
    this.enabled = true;
  }

  initContext() {
    if (!this.ctx && typeof window !== 'undefined') {
      const AudioCtx = window.AudioContext || window.webkitAudioContext;
      if (AudioCtx) this.ctx = new AudioCtx();
    }
    if (this.ctx && this.ctx.state === 'suspended') {
      this.ctx.resume();
    }
  }

  // صوت صفارة الإنذار الأمني عند إرسال بلاغ طوارئ مفقود
  playAlertSound() {
    if (!this.enabled) return;
    this.initContext();
    if (!this.ctx) return;

    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = 'sawtooth';
    // التردد الثنائي المتناوب (880Hz -> 1174Hz)
    osc.frequency.setValueAtTime(880, now);
    osc.frequency.setValueAtTime(1174.66, now + 0.15);
    osc.frequency.setValueAtTime(880, now + 0.3);
    osc.frequency.setValueAtTime(1174.66, now + 0.45);

    gain.gain.setValueAtTime(0.3, now);
    gain.gain.exponentialRampToValueAtTime(0.01, now + 0.6);

    osc.connect(gain);
    gain.connect(this.ctx.destination);
    osc.start(now);
    osc.stop(now + 0.6);
  }

  // نغمة النجاح والعثور على المفتاح (نغمات صاعدة C5 -> C6)
  playSuccessSound() {
    if (!this.enabled) return;
    this.initContext();
    if (!this.ctx) return;

    const now = this.ctx.currentTime;
    const osc = this.ctx.createOscillator();
    const gain = this.ctx.createGain();

    osc.type = 'sine';
    osc.frequency.setValueAtTime(523.25, now);
    osc.frequency.setValueAtTime(659.25, now + 0.1);
    osc.frequency.setValueAtTime(783.99, now + 0.2);
    osc.frequency.setValueAtTime(1046.50, now + 0.3);

    gain.gain.setValueAtTime(0.2, now);
    gain.gain.exponentialRampToValueAtTime(0.001, now + 0.6);

    osc.connect(gain);
    gain.connect(this.ctx.destination);
    osc.start(now);
    osc.stop(now + 0.6);
  }
}

export const sounds = new SoundEffects();

4. عرض بطاقة البلاغ وموقع العثور (src/components/ReportCard.jsx)

يقوم المكون ReportCard بتمثيل حالة البلاغ، وتفعيل بنر خاص بموقع المفتاح فور تسجيله من المكتشف أو رجل الأمن: 

// src/components/ReportCard.jsx (مقتطف البنرات وتثبيت المكان)

export default function ReportCard({ report }) {
  const { setChatReport, activeRole, resolveReportLocation } = useApp();

  const isRoom202 = report.room && report.room.includes('202');

  return (
    <div className={`glass-panel rounded-2xl p-5 ${isRoom202 ? 'ring-1 ring-cyan-400' : ''}`}>
      
      {/* عنوان البلاغ ورقم القاعة */}
      <h3 className="font-extrabold text-white text-base">{report.title}</h3>

      {/* بنر إشعار موقع العثور على المفتاح عند التحديد */}
      {report.foundLocation && (
        <div className="bg-emerald-950/80 p-4 rounded-xl border border-emerald-500/40 mt-3">
          <h4 className="font-extrabold text-xs text-emerald-300">
            📍 أين تم العثور على المفتاح؟
          </h4>
          <p className="text-xs font-bold text-white mt-1">
            {report.foundLocation}
          </p>
          {report.foundBy && (
            <span className="text-[10px] text-emerald-400">
              تم الإيجاد بواسطة: {report.foundBy}
            </span>
          )}
        </div>
      )}

      {/* أزرار التواصل والمتابعة */}
      <button onClick={() => setChatReport(report)} className="btn-primary">
        التواصل والمتابعة 💬
      </button>

    </div>
  );
}

تمكّن نافذة المحادثة المكتشف أو رجل الأمن من إرسال الموقع بدقة وتنبيه المالك وتغيير حالة المفتاح تلقائياً: 

// src/components/ChatModal.jsx

const handleSendMessage = (e) => {
  e.preventDefault();
  if (!text.trim()) return;

  const newMessage = {
    id: Date.now().toString(),
    sender: activeRole === 'security' ? 'رجال الأمن' : 'الطالب/المكتشف',
    text: text.trim(),
    timestamp: new Date().toISOString()
  };

  if (isLocationMsg) {
    // تحديث مكان المفتاح وتغيير حالته إلى (تم العثور عليه)
    resolveReportLocation(chatReport.id, 'found', text.trim(), newMessage.sender);
    sounds.playSuccessSound(); // تشغيل صوت النجاح
    toast.success('تم إرسال الموقع وتحديث حالة البلاغ بنجاح!');
  } else {
    addChatMessage(chatReport.id, newMessage);
    sounds.playMessageSound();
  }

  setText('');
};
